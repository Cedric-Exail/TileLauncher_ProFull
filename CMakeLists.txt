set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets)

target_link_libraries(TileLauncher PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets user32)
